fn main() {
    use std::io::Write;
    use std::process::{Command, Stdio};
    use std::time::{Duration, Instant};
    let exe = std::env::current_exe().unwrap();
    match std::env::args().nth(1).as_deref() {
        Some("--hold") => { std::thread::sleep(Duration::from_secs(5)); println!("HELD_OUTPUT"); return; }
        Some("--late") => { std::thread::sleep(Duration::from_millis(650)); println!("LATE"); return; }
        Some(mode @ ("--root-exits" | "--root-late" | "--root-waits")) => {
            let c = Command::new(&exe).arg(if mode=="--root-late" {"--late"} else {"--hold"})
                .stdin(Stdio::null()).stdout(Stdio::inherit()).stderr(Stdio::inherit()).spawn().unwrap();
            std::fs::write(std::env::var("AC745_REVIEW_PID_PATH").unwrap(), c.id().to_string()).unwrap();
            println!("DESCENDANT_PID={}", c.id());
            if mode=="--root-waits" {std::thread::sleep(Duration::from_secs(5));}
            return;
        }
        Some("--invalid-utf8") => { std::io::stdout().write_all(b"PREFIX\xffSUFFIX").unwrap(); return; }
        _ => {}
    }
    for (mode, ms) in [("--root-exits",500),("--root-late",3000),("--root-waits",500),("--invalid-utf8",1000)] {
        let mut cmd=Command::new(&exe);cmd.arg(mode);
        let started=Instant::now();let run=run_deadlined(cmd,Duration::from_millis(ms)).unwrap();
        println!("CASE={mode} deadline_ms={ms} elapsed_ms={} timeout={} code={} out={:?}",started.elapsed().as_millis(),run.timed_out,run.code,run.stdout);
        if mode!="--invalid-utf8" {
            let pid=std::fs::read_to_string(std::env::var("AC745_REVIEW_PID_PATH").unwrap()).unwrap();
            let alive=Command::new("powershell").args(["-NoProfile","-Command", &format!("if(Get-Process -Id {pid} -ErrorAction SilentlyContinue){{exit 0}}else{{exit 1}}")]).status().unwrap();
            println!("DESCENDANT_ALIVE={}",alive.success());
            if alive.success(){let _=Command::new("taskkill").args(["/PID",&pid,"/T","/F"]).stdout(Stdio::null()).stderr(Stdio::null()).status();}
        }
    }
}
